\section*{Notation}

The $D$-dimensional all-zero and all-one vectors are denoted as $\mathbf{0}_D$ and $\mathbf{1}_D$, respectively. When the dimension of these vectors is given by the context, we omit their dimension and simply write $\mathbf{0}$ and $\mathbf{1}$. The sum of the entries of a vector is denoted as
\begin{equation}
	\sum_{i=0}^{D-1} [\x]_i = \x^\top\mathbf{1}
\end{equation}


$\dict^\top$ --- transpose of matrix $\dict$

$\dict^\mathsf{H}$ --- conjugate transpose of matrix $\dict$

$\dict^\dagger\coloneqq (\dict^\top\dict)^{-1}\dict^\top$ --- pseudo-inverse of matrix $\dict$

$\norm{\X}_2$ --- $l_2$-operator norm of a matrix $\X$

\subsection*{Derivatives}

$\nabla \psi(\x)\in\R^D$ --- gradient of a scalar function $\psi:\R^D\rightarrow\R$ at $\x$, with $[\nabla \psi(\x)]_i = \partial \psi(\x)/\partial x_i$

$\nabla^2 \psi(\x)\in\R^{D\times D}$ --- Hessian of $\psi$ at $\x$, with $[\nabla^2 \psi(\x)]_{ij} = \partial^2 \psi(\x)/\partial x_i\partial x_j$

$\nabla^k \psi(\x)[\v_1,\ldots,\v_k]\in\R$ (for $k\geq 3$) --- the $k$-th derivative of $\psi$ at $\x$ acting as a symmetric $k$-linear form on $\v_1,\ldots,\v_k\in\R^D$, defined as the $k$-th directional derivative
\begin{equation*}
	\nabla^k \psi(\x)[\v_1,\ldots,\v_k] \coloneqq \frac{\partial^k}{\partial s_1\cdots\partial s_k}\bigg|_{s_1=\cdots=s_k=0}\!\!\psi\!\left(\x + \sum_{j=1}^k s_j\v_j\right) = \!\!\sum_{i_1,\ldots,i_k=1}^D\!\frac{\partial^k\psi(\x)}{\partial x_{i_1}\cdots \partial x_{i_k}}[\v_1]_{i_1}\cdots[\v_k]_{i_k}.
\end{equation*}
When all $k$ slots are filled by the same vector $\v$, we use the shorthand $\nabla^k \psi(\x)[\v,\ldots,\v]$. The lower-order cases $k=1$ and $k=2$ are written directly as $\v^\top\nabla\psi(\x)$ and $\v^\top\nabla^2\psi(\x)\v$, respectively.

$\oplus$ --- Orthogonal sum of subspaces

\section{Computation of $\nabla_{\dict} \hat{g}_t(\dict)$}
\label{app:gradient_g_t_dict}

The surrogate function $\hat{g}_t(\dict)$ is given by
\begin{equation}\label{eq:funcionh_app}
	\hat{g}_t(\dict) = \frac{1}{2N_b}\sum\nolimits_{n=1}^{N_b} \norm{ \Pi_\perp(\dict_{\mathcal{A}_{n,t}})\x_{n,t}}^2 
\end{equation} 
where $\Pi_\perp(\dict_{\mathcal{A}_n})= \I -\dict_{\mathcal{A}_n}(\dict^\top_{\mathcal{A}_n}\dict_{\mathcal{A}_n})^{-1}\dict^\top_{\mathcal{A}_n}$. The terms of the sum are of the form
\begin{align}
	\norm{ \Pi_\perp(\dict_{\mathcal{A}})\x}_2^2 &=\Tr(\x\x^\top-(\dict^\top_{\mathcal{A}}\dict_{\mathcal{A}})^{-1}\dict^\top_{\mathcal{A}}\x\x^\top\dict_{\mathcal{A}}) \nonumber %\\
	%&=\Tr\left(\x\x^\top-(\S^\top_\mathcal{A}\dict^\top\dict\S_\mathcal{A})^{-1}\S^\top_\mathcal{A}\dict^\top\x\x^\top\dict\S_\mathcal{A}\right) \nonumber
\end{align}
where the dependence on $n$ and $t$ is dropped for the sake of clarity. The set $\mathcal{A}=\{a_1,\ldots,a_s\}$ where $\{a_l\}_{l=1}^s$ are the indexes of the columns of $\dict$ kept in $\dict_\mathcal{A}$. Let us denote $g(\dict_\mathcal{A})=\Tr(\x\x^\top-(\dict_\mathcal{A}^\top\dict_\mathcal{A})^{-1}\dict_\mathcal{A}^\top \x\x^\top\dict_\mathcal{A})$ and $\dict_\mathcal{A}=\dict\S_\mathcal{A}$, where $\S_\mathcal{A}$ is an $F \times s$ sample matrix whose $l^{th}$ column $\forall l=1,\ldots,s$ is equal to an all zero vector except for entry $(a_l,l)$ which is equal to 1. Using the chain rule, the gradient of $\norm{ \Pi_\perp(\dict_{\mathcal{A}})\x}_2^2$ with respect to $[\dict]_{ij}$ is given by
\begin{equation}\label{eq:gradchainrule}
	\frac{\partial g(\dict_\mathcal{A}(\dict))}{\partial [\dict]_{ij}}=\Tr \left(\left(\nabla_{\dict_\mathcal{A}}g(\dict_\mathcal{A}) \right)^\top\frac{\partial \dict_\mathcal{A}(\dict)}{\partial [\dict]_{ij}}\right)
\end{equation}
Using formula (126) in \cite{matrix_cookbook}, the first term is equal to 
\begin{align}\label{eq:gradgdictA}
	\nabla_{\dict_\mathcal{A}}g(\dict_\mathcal{A})&=-2 \Pi_\perp(\dict_\mathcal{A})\x\x^\top (\dict_\mathcal{A}^\dagger)^\top%=-2\left( \I-\dict_\mathcal{A}(\dict_\mathcal{A}^\top\dict_\mathcal{A})^{-1}\dict_\mathcal{A}^\top\right)\x\x^\top\dict_\mathcal{A}(\dict_\mathcal{A}^\top\dict_\mathcal{A})^{-1} \nonumber \\
	% &=-2 \Pi_\perp(\dict_\mathcal{A})\x\x^\top (\dict_\mathcal{A}^\dagger)^\top
\end{align}
and the second one is equal to 
\begin{equation}\label{eq:graddictAdictij}
	\frac{\partial \dict_\mathcal{A}(\dict)}{\partial [\dict]_{ij}}=\left\{
	\begin{array}{ll}
		\bold{E}_{il} & \textrm{if }j \in \mathcal{A}\\
		\bold{0}_{D\times s} & \textrm{else}
	\end{array} \right.
\end{equation}
where $\bold{E}_{il}$ is an all zeros $D \times s$ matrix except for entry $(i,l)$ with $a_l=j$ that is equal to $1$. Then, substituting \eqref{eq:gradgdictA} and \eqref{eq:graddictAdictij} into \eqref{eq:gradchainrule}, we get
\begin{equation}
	\frac{\partial g(\dict_\mathcal{A}(\dict))}{\partial [\dict]_{ij}}=\left\{
	\begin{array}{ll}
		\left[-2 \Pi_\perp(\dict_\mathcal{A})\x\x^\top (\dict_\mathcal{A}^\dagger)^\top\right]_{il}& \textrm{if }j \in \mathcal{A}\\
		0 & \textrm{else}
	\end{array} \right.
\end{equation}
where $a_l=j$. This leads to
\begin{align}
	\nabla_{\dict}\norm{ \Pi_\perp(\dict_{\mathcal{A}})\x}_2^2&=-2 \Pi_\perp(\dict_\mathcal{A})\x\x^\top (\dict_\mathcal{A}^\dagger)^\top\S^\top_\mathcal{A}
	=-2 (\x - \dict_\mathcal{A}\y_\mathcal{A})\y_\mathcal{A}^\top\S^\top_\mathcal{A} \nonumber \\
	&=-2 (\x - \dict\y)\y^\top \label{eq:gradnorm2}
\end{align}
where $\y_\mathcal{A}\coloneqq \dict_\mathcal{A}^\dagger\x$ and $\y\coloneqq \S_\mathcal{A}\y_\mathcal{A}$. Finally, substituting the gradient \eqref{eq:gradnorm2} with dependence on $n$ and $t$ in each term of the sum in \eqref{eq:funcionh_app} yields
\begin{equation*}
	\nabla_{\dict} \hat{g}_t(\dict) = - \frac{1}{N_b}\sum_n (\x_{n,t} - \dict\y_{n,t})\y_{n,t}^\top 
	= - \frac{1}{N_b} (\X_t - \dict \Y_t) \Y_t^\top
\end{equation*}
where $\X_t\mspace{-4mu}=\mspace{-4mu}[\x_{1,t} \ldots \x_{N_b,t}]$ and $\Y_t\mspace{-4mu}=\mspace{-4mu}[\y_{1,t} \ldots \y_{N_b,t}]$.

\section{Proof of Theorem~\ref{theorem:periodic_graph}}
\label{app:proof_periodic_theorem}

We consider first $P = 2$. Without loss of generality, let us denote by $\lambda^{(rw)}$ any of the eigenvalues of $\L^{(rw)}$. This eigenvalue satisfies that $\det(\L^{(rw)}-\lambda^{(rw)}\I)=0$, which means that the rows of $\L^{(rw)}-\lambda^{(rw)}\I$ denoted by $\{\h_1^\top,\ldots,\h_D^\top\}$ are linearly dependent. Let us now inspect $\L_P^{(rw)}-\lambda^{(rw)}\I$ for $P = 2$
\begin{equation}
	\label{eq:Lrw_P}
	\L_P^{(rw)}-\lambda^{(rw)}\I = \left [
	\begin{array}{ccc}
		\h_1^\top & \qquad\qquad\quad| & \mathbf{0}^\top \\
		\vdots & \qquad\qquad\quad| & \vdots \\
		\h_{D-1}^\top & \qquad\qquad\quad| & \mathbf{0}^\top \\
		\frac{d_D}{d_D+d_1}[\h_D]_{1:D-1}^\top & (1-\lambda^{(rw)}) & \frac{d_1}{d_1+d_D}[\h_1]_{2:D}^\top \\
		\mathbf{0}^\top & |\qquad\qquad\quad & \h_2^\top \\
		\vdots & |\qquad\qquad\quad & \vdots \\
		\mathbf{0}^\top & |\qquad\qquad\quad & \h_D^\top
	\end{array}
	\right ]
\end{equation}
where $\mathbf{0}$ is a zero-vector of length $D - 1$, $d_n$ is the degree of the $n^\text{th}$ vertex of the basic graph $\mathcal{G}$, and $[\h_n]_{l:m}$ is a vector built selecting from entry $l$ to entry $m$ of vector $\h_n$.

As $\{\h_1^\top, \ldots, \h_D^\top\}$ are linearly dependent, there exists a set of scalars $\{\alpha_n\}_{n=1}^D$, where at least one of them different from 0, such that $\sum_{n=1}^D \alpha_n\h_n=\mathbf{0}$. Two cases can be distinguished:
\begin{enumerate}
	\item Both $\alpha_2,\alpha_D\neq0$. In this case, there surely exists a linear combination of $\{\h_1,\ldots,\h_{D-1}\}$ equal to $-\frac{d_D}{d_D+d_1}\h_D$ and another linear combination of $\{\h_2,\ldots,\h_D\}$ equal to $-\frac{d_1}{d_1+d_D}\h_1$. Therefore, there exists a linear combination of the $D - 1$ first and ending rows of $\L_2^{(rw)}-\I$ in~\eqref{eq:Lrw_P} equal to
	\begin{equation}\label{eq:Lrw_Dth_row}
		\left[-\frac{d_D}{d_D+d_1}[\h_D]_{1:D-1}^\top, \beta, -\frac{d_1}{d_1+d_D}[\h_1]_{2:D}^\top\right]
	\end{equation}
	where the term in the middle is equal to
	\begin{equation}\label{eq:Lrw_beta}
		\beta = -\frac{d_D[\h_D]_D + d_1[\h_1]_1}{d_1+d_D}
	\end{equation}
	As per construction of $\L_2^{(rw)}$ we have that $[\h_1]_1=[\h_D]_D=1-\lambda^{(rw)}$, and then $\beta$ in~\eqref{eq:Lrw_beta} is also $-(1-\lambda^{(rw)})$. Therefore, \eqref{eq:Lrw_Dth_row} is equal to the $D^\text{th}$ row in~\eqref{eq:Lrw_P}, which means that the rows of $\L_2^{(rw)}-\lambda^{(rw)}\I$ are linearly dependent, $\det(\L^{(rw)}-\lambda^{(rw)}\I)=0$, and $\lambda^{(rw)}$ is an eigenvalue of $\L_2^{(rw)}$ as well.
	\item At least one of $\{\alpha_1,\ldots,\alpha_D\}$ is equal to 0. In this case, there exists a linear combination of $\{\h_1,\ldots, \h_{D-1}\}$ (and/or $\{\h_2,\ldots, \h_D\}$) that is equal to 0. This means that the
	first (and/or last) $D-1$ rows of the matrix in~\eqref{eq:Lrw_P} are linearly dependent and, therefore, $\det(\L^{(rw)}-\lambda^{(rw)}\I)=0$ and $\lambda^{(rw)}$ is an eigenvalue of $\L_2^{(rw)}$ as well. Note that $\det(\L^{(rw)}-\lambda^{(rw)}\I)=0$ when both $\alpha_1=\alpha_D=0$.
\end{enumerate}
Following the same procedure, it is not difficult to prove by induction that $\lambda^{(rw)}$ is also an eigenvalue of $\L_P^{(rw)}\ \forall P>2$.

\section{Efficient computation of the KKT system in the active-set method}
\label{app:active_set_kkt}

The problem in~\eqref{eq:as_subproblem} can be written compactly as
\begin{equation}
	\lh_\mathcal{W}=\argmin_{\lh\in\R^{D_+}}\tfrac{1}{2}\lh^\top\P\lh-\q^\top\lh
	\quad\text{s.t.}\quad\tilde{\A}_\mathcal{W}\lh=\tilde{\b}_\mathcal{W}.
\end{equation}
Since $\P\succ\mathbf{0}$, this problem admits a closed-form solution via its KKT system
\begin{equation}\label{eq:as_kkt}
	\M_\mathcal{W}
	\begin{bmatrix}
		\lh_\mathcal{W} \\
		\boldsymbol{\nu}_\mathcal{W}
	\end{bmatrix}
	=
	\begin{bmatrix}
		\q \\
		\tilde{\b}_\mathcal{W}
	\end{bmatrix},\quad 
	\text{where}\ \
	\M_\mathcal{W}\coloneqq\begin{bmatrix}
		\P & \tilde{\A}_\mathcal{W}^\top \\
		\tilde{\A}_\mathcal{W} & \mathbf{0}
	\end{bmatrix},
\end{equation}
$\boldsymbol{\nu}_\mathcal{W}\in\R^{D + 1 + |\mathcal{W}|}$ collects the multipliers of the stacked constraints. Because $\P=\beta\D^\top\D$ is diagonal, its inversion costs only $\mathcal{O}(D_+)$. Denoting the KKT matrix in~\eqref{eq:as_kkt} by $\M_\mathcal{W}$, its Schur complement with respect to $\P$ is
\begin{equation}\label{eq:active_set_schur_comp}
	\M_\mathcal{W}|\P = -\tilde{\A}_\mathcal{W}\P^{-1}\tilde{\A}_\mathcal{W}^\top = -\begin{bmatrix}\A\\\B_\mathcal{W}\end{bmatrix}\P^{-1}\begin{bmatrix}\A^\top & \B_\mathcal{W}^\top\end{bmatrix}=  -\begin{bmatrix}\A\P^{-1}\A^\top & \A\P^{-1}\B_\mathcal{W}^\top \\ \B_\mathcal{W}\P^{-1}\A^\top & \B_\mathcal{W}\P^{-1}\B_\mathcal{W}^\top\end{bmatrix}
\end{equation}
which is a $(D+1+|\mathcal{W}|)\times(D+1+|\mathcal{W}|)$ dense matrix in general. The inverse of the KKT block matrix can be written explicitly as
\begin{equation}
	\M_\mathcal{W}^{-1} = 
	\begin{bmatrix}
		\P^{-1}+\P^{-1}\tilde{\A}_\mathcal{W}^\top(\M_\mathcal{W}|\P)^{-1}\tilde{\A}_\mathcal{W}\P^{-1} & -\P^{-1}\tilde{\A}_\mathcal{W}^\top(\M_\mathcal{W}|\P)^{-1} \\
		-(\M_\mathcal{W}|\P)^{-1}\tilde{\A}_\mathcal{W}\P^{-1} & (\M_\mathcal{W}|\P)^{-1}
	\end{bmatrix}.	
\end{equation}
Because $\P$ is diagonal and the constraint matrices $\A$ and $\B$ possess highly structured sparsity patterns, the blocks appearing in the Schur complement admit closed-form expressions and its inverse can be maintained incrementally via rank-1 updates as the working set changes.

\begin{equation}

		\tfrac{1}{2}\C_\mathcal{W} & \mathbf{0}_{D_-}
	\end{bmatrix}
\end{equation}
where $\C_\mathcal{W}$ contains the rows of $\C_1$ indexed by $\mathcal{W}$. 

The rows of $\B_\mathcal{W}$ are mutually exclusive indicator vectors that select off-diagonal entries of $\L$, i.e. each row contains a single entry equal to 1 and all other entries equal to 0, with distinct rows having their non-zero entries in different columns. Therefore, 
\begin{equation}
	\B_\mathcal{W}\P^{-1}\B_\mathcal{W}^\top = \frac{1}{2}\B_\mathcal{W}\B_\mathcal{W}^\top = \frac{1}{2}\I_{|\mathcal{W}|}.
\end{equation}

Using \eqref{eq:BWP_A1} we can write the Schur complement $\mathbf{S}_2$ explicitly:
\begin{equation}
	\begin{split}
	\mathbf{S}_2 &= -\B_\mathcal{W}\P^{-1}\B_\mathcal{W}^\top + \B_\mathcal{W}\P^{-1}\A^\top(\A\P^{-1}\A^\top)^{-1}\A\P^{-1}\B_\mathcal{W}^\top = \\
	&= -\tfrac12\I_{|\mathcal{W}|} + \B_\mathcal{W}\P^{-1}\A^\top\N_{11}^{-1}\A\P^{-1}\B_\mathcal{W}^\top \\
	&= -\tfrac12\I_{|\mathcal{W}|} - \begin{bmatrix} \tfrac12\C_\mathcal{W} & \mathbf{0} \end{bmatrix}
	\begin{bmatrix}
		\displaystyle\frac{2}{D}\I_D-\frac{D-2}{D^2(D-1)}\mathbf{1}_D\mathbf{1}_D^\top &
		\displaystyle-\frac{1}{D(D-1)}\mathbf{1}_D\\[10pt]
		\displaystyle-\frac{1}{D(D-1)}\mathbf{1}_D^\top &
		\displaystyle\frac{1}{D-1}
	\end{bmatrix}
	\begin{bmatrix} \tfrac12\C_\mathcal{W}^\top \\ \mathbf{0}^\top \end{bmatrix} \\
	&= -\tfrac12\I_{|\mathcal{W}|} - \frac{1}{4}\C_\mathcal{W}\Q_{11}\C_\mathcal{W}^\top ,
	\end{split}
\end{equation}
where $\Q_{11}\coloneqq\frac{2}{D}\I_D - \frac{D-2}{D^2(D-1)}\mathbf{1}_D\mathbf{1}_D^\top$. The zero column in \eqref{eq:BWP_A1} annihilates the last row and column of $\N_{11}^{-1}$. And because the last column of $\N_{12}$ (and last row of $\N_{21}$) is zero, the product $\N_{11}^{-1}\N_{12}$ collapses to a single block column. \begin{equation}

		\displaystyle-\frac{1}{D(D-1)}\mathbf{1}_{|\mathcal{W}|}^\top
	\end{bmatrix}.
\end{equation}

	\end{bmatrix}.
\end{equation}

\begin{equation}\label{eq:C_W_add}
\C_{\mathcal{W}_{k+1}} = \begin{bmatrix} \C_{\mathcal{W}_k} \\ \c_{k+1}^\top \end{bmatrix},
\end{equation}
where $\c_{k+1}$ contains exactly two entries equal to $1$ (vertices at the endpoints of $e_{k+1}$) and zeros elsewhere. Conversely, when an edge at position $j$ is removed, $\mathcal{W}_{k+1}=\mathcal{W}_k\setminus\{e_j\}$ and $\C_{\mathcal{W}_{k+1}}$ is obtained by deleting the $j$-th row of $\C_{\mathcal{W}_k}$. 

\subsection{Adding an edge}
\label{subsec:adding_edge}

Substituting the bordered form $\C_{\mathcal{W}_{k+1}}$ from~\eqref{eq:C_W_add} into the explicit expression for $\mathbf{S}_2$ and expanding the quadratic form gives
\begin{equation}
\mathbf{S}_2^{(k+1)} = -\tfrac12\I_{|\mathcal{W}_k|+1} - \frac14\begin{bmatrix} \C_{\mathcal{W}_k}\Q_{11}\C_{\mathcal{W}_k}^\top & \C_{\mathcal{W}_k}\Q_{11}\c_{k+1} \\ \c_{k+1}^\top\Q_{11}\C_{\mathcal{W}_k}^\top & \c_{k+1}^\top\Q_{11}\c_{k+1} \end{bmatrix},
\end{equation}
Because $\mathbf{S}_2^{(k)} = -\tfrac12\I_{|\mathcal{W}_k|} - \frac14\C_{\mathcal{W}_k}\Q_{11}\C_{\mathcal{W}_k}^\top$, this reduces to the bordered matrix
\begin{equation}\label{eq:S2_update}
\mathbf{S}_2^{(k+1)}=
\begin{bmatrix}
\mathbf{S}_2^{(k)} & \u_{k+1}\\[2pt]
\u_{k+1}^\top & \gamma_{k+1}
\end{bmatrix},
\end{equation}
with
\begin{align}
	\label{eq:u_update_vec}
	\u_{k+1} &= -\frac{1}{2D}\C_{\mathcal{W}_k}\c_{k+1} + \frac{D-2}{D^2(D-1)}\mathbf{1}_{|\mathcal{W}_k|},\\
	\label{eq:gamma_update}
	\gamma_{k+1} &= -\frac12 - \frac{1}{2D}\c_{k+1}^\top\c_{k+1} + \frac{D-2}{D^2(D-1)} = \frac{-D^3-D^2+4D-4}{2D^2(D-1)}.
\end{align}
Since $\c_{k+1}^\top\c_{k+1}=2$ and $\c_i^\top\c_{k+1}$ with $i\neq k+1$ equals $1$ when $e_i$ shares a vertex with $e_{k+1}$ and $0$ otherwise, the $i^\text{th}$ entry of $\u_{k+1}$ becomes
\begin{equation}
	\label{eq:u_entrywise_derived}
	[\u_{k+1}]_i=
	\begin{cases}
		\displaystyle\frac{-D^2+3D-4}{2D^2(D-1)}, & e_i\sim e_{k+1},\\[8pt]
		\displaystyle\frac{D-2}{D^2(D-1)}, & e_i\not\sim e_{k+1},
	\end{cases}
	\qquad i=1,\dots,|\mathcal{W}_k|.
\end{equation}
For the very first edge there is no previous working set, so $\u_1$ is empty and $\mathbf{S}_2^{(1)}=[\gamma_1]$ with $\gamma_1=\frac{-D^3-D^2+4D-4}{2D^2(D-1)}$. Using the block-inversion formula, the inverse of the bordered matrix is obtained from $(\mathbf{S}_2^{(k)})^{-1}$ as
\begin{equation}\label{eq:S2_inv_update}
(\mathbf{S}_2^{(k+1)})^{-1} = \begin{bmatrix} (\mathbf{S}_2^{(k)})^{-1} + \frac{1}{\sigma_{k+1}}\v_{k+1}\v_{k+1}^\top & -\frac{1}{\sigma_{k+1}}\v_{k+1} \\[2pt] -\frac{1}{\sigma_{k+1}}\v_{k+1}^\top & \frac{1}{\sigma_{k+1}} \end{bmatrix},
\end{equation}
where $\v_{k+1}\coloneqq(\mathbf{S}_2^{(k)})^{-1}\u_{k+1}$ and $\sigma_{k+1}\coloneqq\gamma_{k+1}-\u_{k+1}^\top\v_{k+1}=\gamma_{k+1}-\u_{k+1}^\top(\mathbf{S}_2^{(k)})^{-1}\u_{k+1}$ is the scalar Schur complement. Consequently $(\mathbf{S}_2^{(k)})^{-1}$ can be maintained with rank-1 updates (adding) or downdates (removing) at cost $\mathcal{O}(|\mathcal{W}|^2)$ per iteration, avoiding the $\mathcal{O}(|\mathcal{W}|^3)$ expense of recomputing the inverse from scratch. Similarly, the matrix $\mathbf{S}_3$ is updated by appending the column
\begin{equation}
	\label{eq:s_e}
	\s_{k+1}
	=
	\begin{bmatrix}
		\displaystyle\frac{1}{D}\c_{k+1}-\frac{D-2}{D^2(D-1)}\mathbf{1}_D\\[10pt]
		\displaystyle-\frac{1}{D(D-1)}
	\end{bmatrix},
\end{equation}
so that
\begin{equation}
	\label{eq:S3_update}
	\mathbf{S}_3^{(k+1)}
	=
	\begin{bmatrix}
		\mathbf{S}_3^{(k)} & \s_{k+1}
	\end{bmatrix}.
\end{equation}
This avoids recomputing the full Schur-complement inverse from scratch and keeps the cost at $\mathcal{O}((D+|\mathcal{W}|)^2)$ per working-set addition.

Using the bordered inverse of $\mathbf{S}_2$ and the column update of $\mathbf{S}_3$, the blocks of \eqref{eq:MWP_inv_compact} update as follows. Let
\begin{equation}
	\label{eq:K_blocks}
	(\M_{\mathcal{W}_k}|\P)^{-1}
	=
	-\begin{bmatrix}
		\mathbf{K}_{11}^{(k)} & \mathbf{K}_{12}^{(k)} \\[4pt]
		\mathbf{K}_{21}^{(k)} & \mathbf{K}_{22}^{(k)}
	\end{bmatrix},
\end{equation}
where, from \eqref{eq:MWP_inv_compact},
\begin{equation}
	\label{eq:K_blocks_def}
	\mathbf{K}_{11}^{(k)}=\N_{11}^{-1}+\mathbf{S}_3^{(k)}(\mathbf{S}_2^{(k)})^{-1}(\mathbf{S}_3^{(k)})^\top,\quad
	\mathbf{K}_{12}^{(k)}=-\mathbf{S}_3^{(k)}(\mathbf{S}_2^{(k)})^{-1},\quad
	\mathbf{K}_{22}^{(k)}=(\mathbf{S}_2^{(k)})^{-1}.
\end{equation}
To update \eqref{eq:K_blocks} from step $k$ to $k+1$, substitute the bordered updates of $\mathbf{S}_3$ and $\mathbf{S}_2^{-1}$. The off-diagonal blocks are updated as
\begin{align}
	\K_{12}^{(k+1)}=-\mathbf{S}_3^{(k+1)}(\mathbf{S}_2^{(k+1)})^{-1}
	&=
	-\begin{bmatrix} \mathbf{S}_3^{(k)} & \s_e^{(k+1)} \end{bmatrix}
	\begin{bmatrix}
		(\mathbf{S}_2^{(k)})^{-1}+\frac{1}{\sigma_{k+1}}\v_{k+1}\v_{k+1}^\top & -\frac{1}{\sigma_{k+1}}\v_{k+1} \\[4pt]
		-\frac{1}{\sigma_{k+1}}\v_{k+1}^\top & \frac{1}{\sigma_{k+1}}
	\end{bmatrix} \nonumber \\
	&=
	-\begin{bmatrix}
		\mathbf{S}_3^{(k)}(\mathbf{S}_2^{(k)})^{-1}+\frac{1}{\sigma_{k+1}}\z_{k+1}\v_{k+1}^\top &
		-\frac{1}{\sigma_{k+1}}\z_{k+1}
	\end{bmatrix}= \nonumber \\
	&=\begin{bmatrix}
		\K_{12}^{(k)}-\frac{1}{\sigma_{k+1}}\z_{k+1}\v_{k+1}^\top &
		\frac{1}{\sigma_{k+1}}\z_{k+1}
	\end{bmatrix}
\end{align}
where $\z_{k+1}\coloneqq \mathbf{S}_3^{(k)}\v_{k+1}-\s_e^{(k+1)}$ is an auxiliary vector. With this definition, the top-left block of \eqref{eq:MWP_inv_compact} is updated as
\begin{align}
	\K_{11}^{(k+1)}&=\N_{11}^{-1}+\mathbf{S}_3^{(k+1)}(\mathbf{S}_2^{(k+1)})^{-1}(\mathbf{S}_3^{(k+1)})^\top
	=\nonumber\\
	&=\N_{11}^{-1}+\begin{bmatrix}
		\mathbf{S}_3^{(k)}(\mathbf{S}_2^{(k)})^{-1}+\frac{1}{\sigma_{k+1}}\z_{k+1}\v_{k+1}^\top &
		-\frac{1}{\sigma_{k+1}}\z_{k+1}
	\end{bmatrix}
	\begin{bmatrix} (\mathbf{S}_3^{(k)})^\top \\ \s_{k+1}^\top \end{bmatrix}= \nonumber \\
	&= \N_{11}^{-1}+\mathbf{S}_3^{(k)}(\mathbf{S}_2^{(k)})^{-1}(\mathbf{S}_3^{(k)})^\top+\frac{1}{\sigma_{k+1}}\z_{k+1}\z_{k+1}^\top=\K_{11}^{(k)}+\frac{1}{\sigma_{k+1}}\z_{k+1}\z_{k+1}^\top.
\end{align}

Define the weighted sum of active edge incidence vectors $\mathbf{w}_{k+1}\coloneqq\C_{\mathcal{W}_k}^\top\v_{k+1}\in\mathbb{R}^D$ and the scalar $s_{k+1}\coloneqq\mathbf{1}_{|\mathcal{W}_k|}^\top\v_{k+1}$. The vector $\z_{k+1}$ follows without forming $\S_3^{(k)}$. From \eqref{eq:U_def},
\begin{equation}
	\S_3^{(k)}\v_{k+1}
	=
	\begin{bmatrix}
		\displaystyle\frac{1}{D}\mathbf{w}_{k+1} - \frac{D-2}{D^2(D-1)}s_{k+1}\mathbf{1}_D\\[10pt]
		\displaystyle -\frac{1}{D(D-1)}s_{k+1}
	\end{bmatrix}.
\end{equation}
Subtracting the new column $\s_{k+1}$ from \eqref{eq:s_e} gives
\begin{equation}
	\label{eq:z_update_vec}
	\z_{k+1}
	=
	\S_3^{(k)}\v_{k+1}-\s_{k+1}
	=
	\begin{bmatrix}
		\displaystyle\frac{1}{D}(\mathbf{w}_{k+1}-\c_{k+1}) - \frac{D-2}{D^2(D-1)}(s_{k+1}-1)\mathbf{1}_D\\[10pt]
		\displaystyle -\frac{1}{D(D-1)}(s_{k+1}-1)
	\end{bmatrix}.
\end{equation}
Consequently $\S_3^{(k)}$ itself never needs to be stored. The only edge-related object that must be maintained is the active incidence matrix $\C_{\mathcal{W}_k}$ (equivalently, the list of active edges); from it, $\mathbf{w}_{k+1}=\C_{\mathcal{W}_k}^\top\v_{k+1}$ and $s_{k+1}=\mathbf{1}_{|\mathcal{W}_k|}^\top\v_{k+1}$ are formed on the fly for each new edge, and \eqref{eq:z_update_vec} produces $\z_{k+1}$ directly.


Finally, the bottom-right block of \eqref{eq:MWP_inv_compact} is $\mathbf{K}_{22}^{(k+1)}=(\mathbf{S}_2^{(k+1)})^{-1}$, which is obtained directly from the block inversion of \eqref{eq:S2_inv_update}
\begin{equation}
(\K_{22}^{(k+1)})^{-1} = \begin{bmatrix} (\K_{22}^{(k)})^{-1} + \frac{1}{\sigma_{k+1}}\v_{k+1}\v_{k+1}^\top & -\frac{1}{\sigma_{k+1}}\v_{k+1} \\[2pt] -\frac{1}{\sigma_{k+1}}\v_{k+1}^\top & \frac{1}{\sigma_{k+1}} \end{bmatrix}.
\end{equation}
As a result, the four blocks of \eqref{eq:MWP_inv_compact} become
\begin{equation}
	\label{eq:MWP_inv_update}
	(\M_{\mathcal{W}_{k+1}}|\P)^{-1}
	=
	-\begin{bmatrix}
		\mathbf{K}_{11}^{(k)}+\frac{1}{\sigma_{k+1}}\z_{k+1}\z_{k+1}^\top &
		\mathbf{K}_{12}^{(k)}-\frac{1}{\sigma_{k+1}}\z_{k+1}\v_{k+1}^\top &
		\frac{1}{\sigma_{k+1}}\z_{k+1} \\[6pt]
		\mathbf{K}_{21}^{(k)}-\frac{1}{\sigma_{k+1}}\v_{k+1}\z_{k+1}^\top &
		\mathbf{K}_{22}^{(k)}+\frac{1}{\sigma_{k+1}}\v_{k+1}\v_{k+1}^\top &
		-\frac{1}{\sigma_{k+1}}\v_{k+1} \\[6pt]
		\frac{1}{\sigma_{k+1}}\z_{k+1}^\top &
		-\frac{1}{\sigma_{k+1}}\v_{k+1}^\top &
		\frac{1}{\sigma_{k+1}}
	\end{bmatrix}.
\end{equation}
Equivalently, as a single rank-1 downdate of the bordered old inverse,
\begin{equation}
	\label{eq:MWP_inv_update_rank1}
	(\M_{\mathcal{W}_{k+1}}|\P)^{-1}
	=
	\begin{bmatrix}
		(\M_{\mathcal{W}_{k}}|\P)^{-1} & \mathbf{0} \\[2pt]
		\mathbf{0}^\top & 0
	\end{bmatrix}
	-
	\frac{1}{\sigma_{k+1}}
	\begin{bmatrix}
		\z_{k+1} \\[2pt] -\v_{k+1} \\[2pt] 1
	\end{bmatrix}
	\begin{bmatrix}
		\z_{k+1}^\top & -\v_{k+1}^\top & 1
	\end{bmatrix}.
\end{equation}
Recall from \eqref{eq:as_kkt} that the KKT system we solve at each active-set iteration is
\begin{equation}
	\M_\mathcal{W}
	\begin{bmatrix}
		\lh_\mathcal{W} \\
		\boldsymbol{\nu}_\mathcal{W}
	\end{bmatrix}
	=
	\begin{bmatrix}
		\q \\
		\tilde{\b}_\mathcal{W}
	\end{bmatrix},\quad
	\text{where }\ 
	\M_\mathcal{W}\coloneqq\begin{bmatrix}
		\P & \tilde{\A}_\mathcal{W}^\top \\
		\tilde{\A}_\mathcal{W} & \mathbf{0}
	\end{bmatrix}.
\end{equation}
The block inverse of the KKT matrix is
\begin{equation}
	\label{eq:MW_full_inv}
	\M_\mathcal{W}^{-1}
	=
	\begin{bmatrix}
		\P^{-1}+\P^{-1}\tilde{\A}_\mathcal{W}^\top(\M_\mathcal{W}|\P)^{-1}\tilde{\A}_\mathcal{W}\P^{-1} & -\P^{-1}\tilde{\A}_\mathcal{W}^\top(\M_\mathcal{W}|\P)^{-1} \\[6pt]
		-(\M_\mathcal{W}|\P)^{-1}\tilde{\A}_\mathcal{W}\P^{-1} & (\M_\mathcal{W}|\P)^{-1}
	\end{bmatrix}.
\end{equation}
The two block rows of \eqref{eq:as_kkt} read
\begin{equation}
	\label{eq:kkt_primal_block}
	\P\lh_\mathcal{W}+\tilde{\A}_\mathcal{W}^\top\boldsymbol{\nu}_\mathcal{W}=\q,
	\qquad
	\tilde{\A}_\mathcal{W}\lh_\mathcal{W}=\tilde{\b}_\mathcal{W}.
\end{equation}
Eliminating $\lh_\mathcal{W}$ from the first equation and substituting into the second yields
\begin{equation}
	\tilde{\A}_\mathcal{W}\P^{-1}\tilde{\A}_\mathcal{W}^\top\boldsymbol{\nu}_\mathcal{W}
	=
	\tilde{\A}_\mathcal{W}\P^{-1}\q-\tilde{\b}_\mathcal{W}.
\end{equation}
Writing $\mathbf{r}\coloneqq\tilde{\A}_\mathcal{W}\P^{-1}\q - \tilde{\b}_\mathcal{W}$ and recalling that the Schur complement is $\M_\mathcal{W}|\P=-\tilde{\A}_\mathcal{W}\P^{-1}\tilde{\A}_\mathcal{W}^\top$, the dual variables satisfy
\begin{equation}
	\label{eq:dual_schur_system}
	(\M_\mathcal{W}|\P)\boldsymbol{\nu}_\mathcal{W}=-\mathbf{r},
	\qquad\text{so}\qquad
	\boldsymbol{\nu}_\mathcal{W}=-(\M_\mathcal{W}|\P)^{-1}\mathbf{r}.
\end{equation}
The rank-1 update \eqref{eq:MWP_inv_update_rank1} gives an immediate update formula when an edge is added to the working set. The residual vector is bordered as
\begin{equation}
	\label{eq:r_update}
	\mathbf{r}^{(k+1)}
	\coloneqq
	\tilde{\A}_{\mathcal{W}_{k+1}}\P^{-1}\q - \tilde{\b}_{\mathcal{W}_{k+1}}
	=
	\begin{bmatrix}
		\mathbf{r}^{(k)} \\ r_{k+1}
	\end{bmatrix},
	\qquad
	r_{k+1}\coloneqq \tilde{\a}_{k+1}^\top\P^{-1}\q - \tilde{b}_{k+1},
\end{equation}
where $\tilde{\a}_{k+1}^\top$ and $\tilde{b}_{k+1}$ are the new row of $\tilde{\A}_{\mathcal{W}_{k+1}}$ and the corresponding entry of $\tilde{\b}_{\mathcal{W}_{k+1}}$, respectively. Substituting \eqref{eq:MWP_inv_update_rank1} and \eqref{eq:r_update} into $\boldsymbol{\nu}_{\mathcal{W}_{k+1}}=-(\M_{\mathcal{W}_{k+1}}|\P)^{-1}\mathbf{r}^{(k+1)}$ gives
\begin{multline}
	\boldsymbol{\nu}_{\mathcal{W}_{k+1}}
	=
	-
	\begin{bmatrix}
		(\M_{\mathcal{W}_{k}}|\P)^{-1} & \mathbf{0} \\
		\mathbf{0}^\top & 0
	\end{bmatrix}
	\begin{bmatrix}
		\mathbf{r}^{(k)} \\ r_{k+1}
	\end{bmatrix}
	+
	\frac{1}{\sigma_{k+1}}
	\begin{bmatrix}
		\z_{k+1} \\ -\v_{k+1} \\ 1
	\end{bmatrix}
	\begin{bmatrix}
		\z_{k+1}^\top & -\v_{k+1}^\top & 1
	\end{bmatrix}
	\begin{bmatrix}
		\mathbf{r}^{(k)} \\ r_{k+1}
	\end{bmatrix}.
\end{multline}
The first term is the old dual vector padded with a zero, because $\boldsymbol{\nu}_{\mathcal{W}_k}=-(\M_{\mathcal{W}_{k}}|\P)^{-1}\mathbf{r}^{(k)}$. Define the scalar
\begin{equation}
	\label{eq:tau_update}
	\tau_{k+1}
	\coloneqq
	\begin{bmatrix}
		\z_{k+1}^\top & -\v_{k+1}^\top & 1
	\end{bmatrix}
	\mathbf{r}^{(k+1)}
	=
	\z_{k+1}^\top\mathbf{r}_1 - \v_{k+1}^\top\mathbf{r}_2^{(k)} + r_{k+1}.
\end{equation}
Therefore the dual variables update by the rank-1 correction
\begin{equation}
	\label{eq:nu_update}
	\boldsymbol{\nu}_{\mathcal{W}_{k+1}}
	=
	\begin{bmatrix}
		\boldsymbol{\nu}_{\mathcal{W}_k} \\ 0
	\end{bmatrix}
	+
	\frac{\tau_{k+1}}{\sigma_{k+1}}
	\begin{bmatrix}
		\z_{k+1} \\ -\v_{k+1} \\ 1
	\end{bmatrix}.
\end{equation}
Partitioning $\boldsymbol{\nu}_{\mathcal{W}_{k+1}}=[\boldsymbol{\mu}_{k+1}^\top,\boldsymbol{\eta}_{\mathcal{W}_{k+1}}^\top]^\top$, this is equivalent to
\begin{equation}
	\label{eq:eta_mu_update_direct}
	\boldsymbol{\mu}_{k+1}
	=
	\boldsymbol{\mu}_k + \frac{\tau_{k+1}}{\sigma_{k+1}}\z_{k+1},
	\qquad
	\boldsymbol{\eta}_{\mathcal{W}_{k+1}}
	=
	\begin{bmatrix}
		\boldsymbol{\eta}_{\mathcal{W}_k} \\ 0
	\end{bmatrix}
	-
	\frac{\tau_{k+1}}{\sigma_{k+1}}
	\begin{bmatrix}
		\v_{k+1} \\ -1
	\end{bmatrix}.
\end{equation}

Finally, the primal iterate follows from the first block row of \eqref{eq:as_kkt}. Eliminating $\lh_\mathcal{W}$ from \eqref{eq:kkt_primal_block} gives
\begin{equation}
	\label{eq:lambda_from_nu}
	\lh_{\mathcal{W}}
	=
	\P^{-1}\q - \P^{-1}\tilde{\A}_{\mathcal{W}}^\top\boldsymbol{\nu}_{\mathcal{W}}.
\end{equation}
Substituting \eqref{eq:nu_update} yields
\begin{equation}
	\label{eq:lambda_update}
	\lh_{\mathcal{W}_{k+1}}
	=
	\lh_{\mathcal{W}_k}
	-
	\frac{\tau_{k+1}}{\sigma_{k+1}}\P^{-1}\tilde{\A}_{\mathcal{W}_{k+1}}^\top
	\begin{bmatrix}
		\z_{k+1} \\ -\v_{k+1} \\ 1
	\end{bmatrix}.
\end{equation}

\subsection{Removing an edge}
\label{subsec:removing_edge}

Let the current working set be $\mathcal{W}_k$, with $|\mathcal{W}_k|=m$, and let $j\in\{1,\dots,m\}$ be the position of the edge that is dropped. Write $\mathcal{W}_{k+1}\coloneqq\mathcal{W}_k\setminus\{e_j\}$, and let $\c_{k+1}\in\{0,1\}^D$ be the incidence vector of the removed edge (the $j$-th row of $\C_{\mathcal{W}_k}$). Let $\mathbf{P}_j\in\{0,1\}^{m\times m}$ be the permutation matrix that moves the $j$-th active edge to the last position. Then the current Schur-complement matrix can be written in the same bordered form used for additions,
\begin{equation}
	\label{eq:S2_border_remove}
	\mathbf{P}_j\mathbf{S}_2^{(k)}\mathbf{P}_j^\top
	=
	\begin{bmatrix}
		\mathbf{S}_2^{(k+1)} & \u_{k+1} \\
		\u_{k+1}^\top & \gamma_{k+1}
	\end{bmatrix},
\end{equation}
where $\mathbf{S}_2^{(k+1)}\in\R^{(m-1)\times(m-1)}$ is the Schur complement after removal and $\u_{k+1}\in\R^{m-1}$, $\gamma_{k+1}\in\R$ are the column and diagonal entry associated with the leaving edge. All the information needed to recover $(\mathbf{S}_2^{(k+1)})^{-1}$ is contained in the maintained inverse $(\mathbf{S}_2^{(k)})^{-1}$. To see this, write the Schur-complement inverse of the bordered matrix~
\eqref{eq:S2_border_remove}:
\begin{equation}
	\mathbf{P}_j(\mathbf{S}_2^{(k)})^{-1}\mathbf{P}_j^\top
	=
	\begin{bmatrix}
		(\mathbf{S}_2^{(k+1)})^{-1}+\frac{1}{\sigma_{k+1}}(\mathbf{S}_2^{(k+1)})^{-1}\u_{k+1}\u_{k+1}^\top(\mathbf{S}_2^{(k+1)})^{-1} & -\frac{1}{\sigma_{k+1}}(\mathbf{S}_2^{(k+1)})^{-1}\u_{k+1} \\[6pt]
		-\frac{1}{\sigma_{k+1}}\u_{k+1}^\top(\mathbf{S}_2^{(k+1)})^{-1} & \frac{1}{\sigma_{k+1}}
	\end{bmatrix},
\end{equation}
where $\sigma_{k+1}\coloneqq\gamma_{k+1}-\u_{k+1}^\top(\mathbf{S}_2^{(k+1)})^{-1}\u_{k+1}$ is the scalar Schur complement. The bottom-right entry and the off-diagonal column of the permuted maintained inverse are therefore
\begin{equation}
	d_{k+1}\coloneqq[\mathbf{P}_j(\mathbf{S}_2^{(k)})^{-1}\mathbf{P}_j^\top]_{mm}=\frac{1}{\sigma_{k+1}},
	\
	\tilde{\v}_{k+1}\coloneqq(\mathbf{P}_j(\mathbf{S}_2^{(k)})^{-1}\mathbf{P}_j^\top)_{\setminus m,m}=-\frac{1}{\sigma_{k+1}}(\mathbf{S}_2^{(k+1)})^{-1}\u_{k+1}.
\end{equation}
Because $\mathbf{P}_j$ only reorders rows and columns, these are exactly the $j$-th diagonal entry and the $j$-th off-diagonal column of $(\mathbf{S}_2^{(k)})^{-1}$; equivalently,
\begin{equation}
	d_{k+1}=[(\mathbf{S}_2^{(k)})^{-1}]_{jj},
	\qquad
	\tilde{\v}_{k+1}=((\mathbf{S}_2^{(k)})^{-1})_{\setminus j,j}.
\end{equation}
Solving for $(\mathbf{S}_2^{(k+1)})^{-1}\u_{k+1}$ gives the auxiliary vector
\begin{equation}
	\label{eq:v_downdate}
	\v_{k+1}\coloneqq(\mathbf{S}_2^{(k+1)})^{-1}\u_{k+1}=-\frac{1}{d_{k+1}}\tilde{\v}_{k+1}.
\end{equation}
Finally, the top-left block of the maintained inverse reads
\begin{equation}
	((\mathbf{S}_2^{(k)})^{-1})_{\setminus j,\setminus j}
	=
	(\mathbf{S}_2^{(k+1)})^{-1}+\frac{1}{\sigma_{k+1}}\v_{k+1}\v_{k+1}^\top,
\end{equation}
so isolating $(\mathbf{S}_2^{(k+1)})^{-1}$ yields
\begin{equation}
	\label{eq:S2_inv_downdate}
	(\mathbf{S}_2^{(k+1)})^{-1}
	=
	((\mathbf{S}_2^{(k)})^{-1})_{\setminus j,\setminus j}
	-
	\frac{1}{d_{k+1}}\tilde{\v}_{k+1}\tilde{\v}_{k+1}^\top.
\end{equation}

Next, let $\C^{(k+1)}$ denote the active incidence matrix $\C_{\mathcal{W}_k}$ with row $j$ removed, and define
\begin{equation}
	\w_{k+1}\coloneqq(\C^{(k+1)})^\top\v_{k+1},
	\qquad
	s_{k+1}\coloneqq\mathbf{1}_{m-1}^\top\v_{k+1}.
\end{equation}
The vector $\z$ for the downdate is computed from these quantities and from the incidence vector $\c_{k+1}$ of the removed edge,
\begin{equation}
	\label{eq:z_downdate_remove}
	\z_{k+1}
	=
	\begin{bmatrix}
		\displaystyle\frac{1}{D}(\w_{k+1}-\c_{k+1})-\frac{D-2}{D^2(D-1)}(s_{k+1}-1)\mathbf{1}_D \\[10pt]
		\displaystyle-\frac{1}{D(D-1)}(s_{k+1}-1)
	\end{bmatrix}.
\end{equation}
Equivalently, $\z_{k+1}=\mathbf{S}_3^{(k+1)}\v_{k+1}-\s_{k+1}$, where $\mathbf{S}_3^{(k+1)}$ is the reduced $\mathbf{S}_3$ matrix and $\s_{k+1}$ is the column of $\mathbf{S}_3^{(k)}$ associated with the removed edge.

Because the current inverse $(\M_{\mathcal{W}_k}|\P)^{-1}$ was obtained from $(\M_{\mathcal{W}_{k+1}}|\P)^{-1}$ by the rank-1 update in~\eqref{eq:MWP_inv_update_rank1}, removing the edge amounts to reversing that correction. The multiplier of the leaving edge is the $(D+1+j)$-th entry of the current dual vector,
\begin{equation}
	\alpha_{k+1}\coloneqq[\boldsymbol{\nu}_{\mathcal{W}_k}]_{D+1+j}.
\end{equation}
Reversing the rank-1 correction gives the reduced dual vector
\begin{equation}
	\label{eq:nu_downdate}
	\boldsymbol{\nu}_{\mathcal{W}_{k+1}}
	=
	(\boldsymbol{\nu}_{\mathcal{W}_k})_{\setminus(D+1+j)}
	-
	\alpha_{k+1}
	\begin{bmatrix}
		\z_{k+1} \\ -\v_{k+1}
	\end{bmatrix}.
\end{equation}
Finally, using $\lh_{\mathcal{W}_k}=\P^{-1}\q-\P^{-1}\tilde{\A}_{\mathcal{W}_k}^\top\boldsymbol{\nu}_{\mathcal{W}_k}$, the primal iterate is updated as
\begin{equation}
	\label{eq:lambda_downdate}
	\lh_{\mathcal{W}_{k+1}}
	=
	\lh_{\mathcal{W}_k}
	+
	\alpha_{k+1}
	\P^{-1}
	\tilde{\A}_{\mathcal{W}_k}^\top
	\begin{bmatrix}
		\z_{k+1} \\ -\v_{k+1} \\ 1
	\end{bmatrix},
\end{equation}
and the active residual is shortened by deleting the $j$-th entry, $\r_2^{(k+1)}=(\r_2^{(k)})_{\setminus j}$.

The arithmetic cost of a removal is again dominated by the rank-1 downdate of $(\mathbf{S}_2^{(k)})^{-1}$ and by the matrix--vector products involving $\C^{(k+1)}$, so it is $\mathcal{O}((D+|\mathcal{W}_k|)^2)$. Algorithm~\ref{alg:active_set_remove_edge} collects these steps in pseudocode.

\begin{remark}
	Note that the dual variables $\boldsymbol{\mu}$ and $\boldsymbol{\eta}_\mathcal{W}$ are not computed merely as an intermediate step for obtaining $\lh_\mathcal{W}$. In an active-set method they are essential: their signs govern which constraints enter or leave the working set, and they are required for the optimality test at each iteration. Maintaining the full primal-dual pair $(\lh_\mathcal{W},\boldsymbol{\nu}_\mathcal{W})$ is therefore unavoidable, and the rank-1 updates \eqref{eq:nu_update} and \eqref{eq:nu_downdate} reuse the duals at no extra cost.
\end{remark}

The rank-1 update formulas derived above are used by the active-set iteration in Section~\ref{sec:active_set_method}; the corresponding pseudocode is given in Algorithms~\ref{alg:active_set_add_edge} and~\ref{alg:active_set_remove_edge}.
